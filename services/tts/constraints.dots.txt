# Constraints for the dots.tts engine build.
#
# gfx1201 (Radeon R9700 AI Pro) needs ROCm 7.x, and the only torch/torchaudio
# matched pair on the pytorch.org rocm7.2 index is 2.11.0 — see Dockerfile.dots.
# These pins freeze the torch stack so that when the dots.tts transitive deps are
# installed from PyPI (+ rocm7.2 extra-index), pip cannot "upgrade" torch to a
# CUDA build from PyPI and silently replace the ROCm runtime. The local-version
# segment (+rocm7.2) is required for pip's == to match the installed wheel.
torch==2.11.0+rocm7.2
torchaudio==2.11.0+rocm7.2

# Match dots.tts' recommended.txt for the heavier scientific deps.
numpy==2.2.6
transformers==4.57.0
safetensors==0.8.0rc0
librosa==0.11.0
soundfile==0.13.1
pydantic==2.12.5
PyYAML==6.0.3
